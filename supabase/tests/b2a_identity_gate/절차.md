  from pg_catalog.pg_namespace n
 where n.nspname <> 'security_private'
   and n.nspname not like 'pg\_toast%' and n.nspname not like 'pg\_temp%';

select pg_catalog.md5(coalesce(pg_catalog.string_agg(
         d.defaclrole::text || '|' || d.defaclnamespace::text || '|' || d.defaclobjtype::text || '|' || d.defaclacl::text,
         E'\n' order by d.defaclrole, d.defaclnamespace, d.defaclobjtype), '')) as default_acl_fp
  from pg_catalog.pg_default_acl d;

-- (바) 사람이 만든 트리거(외래키가 만드는 내부 트리거는 뺀다 — 마이그레이션이 auth.users·public.users 에
--      내부 트리거를 붙이고, 되돌리면 사라진다)
select pg_catalog.count(*) as n,
       pg_catalog.md5(coalesce(pg_catalog.string_agg(
         t.tgrelid::pg_catalog.regclass::text || '|' || t.tgname || '|' || t.tgenabled::text || '|' || t.tgfoid::pg_catalog.regproc::text,
         E'\n' order by t.tgrelid::pg_catalog.regclass::text, t.tgname), '')) as fp
  from pg_catalog.pg_trigger t
 where not t.tgisinternal
   and t.tgrelid::pg_catalog.regclass::text not like '%zz\_b2a\_%';

-- (사) 로그인 계정·로그인 상태 수(시험 뒤 합성 계정을 다 지웠는지 보려는 기준)
select (select pg_catalog.count(*) from auth.users) as auth_users,
       (select pg_catalog.count(*) from auth.sessions) as auth_sessions,
       (select pg_catalog.count(*) from public.users) as public_users;
```
- 이 질의들은 작업방이 돌려 보지 않았다. 1차 실행에서 브레인방이 전부 실행했다(문법 오류 없음). 2차에서 질의는 바꾸지 않았다.
- 로그인 서버가 자기 표를 스스로 바꾸는 경우(예: 내부 업데이트)가 있으면 (가)~(라)가 이 작업과 무관하게 달라질 수 있다. 달라지면 어느 스키마의 무엇이 달라졌는지 풀어서 확인한다.

insert into security_private.access_grant (credential_digest, purpose, target_member_id, issued_at, expires_at)
values (pg_catalog.encode(pg_catalog.sha256('zz-b2a-edge-exp'::bytea), 'hex'), 'book', '<A_회원번호>',
        pg_catalog.now() - interval '10 minutes', pg_catalog.now());            -- 만료 시각 = 지금
insert into security_private.access_grant (credential_digest, purpose, target_member_id, issued_at, expires_at)
values (pg_catalog.encode(pg_catalog.sha256('zz-b2a-edge-iss'::bytea), 'hex'), 'book', '<A_회원번호>',
        pg_catalog.now(), pg_catalog.now() + interval '10 minutes');            -- 발급 시각 = 지금
select pg_catalog.count(*) as expect_0 from security_private.resolve_grant(
  pg_catalog.encode(pg_catalog.sha256('zz-b2a-edge-exp'::bytea), 'hex'), 'book_read');
select pg_catalog.count(*) as expect_1 from security_private.resolve_grant(
  pg_catalog.encode(pg_catalog.sha256('zz-b2a-edge-iss'::bytea), 'hex'), 'book_read');
rollback;
```
| 보낼 요청 | 기대 응답 | 변화 |
|---|---|---|
| 위 묶음 | `expect_0` = 0, `expect_1` = 1 | 없음(rollback) |

|---|---|---|
| `purpose='proxy'`, `actor_auth_user_id = null`, 20분 | `23514` (`access_grant_proxy_check`) | 0 |
| `purpose='proxy'`, 행위자 G, `expires_at = issued_at + interval '30 minutes 1 second'` | `23514` (`access_grant_proxy_check`) | 0 |
| `purpose='proxy'`, 행위자 G, `issued_at = T`, `expires_at = T + interval '30 minutes'`(정확히 30분) — `begin; … rollback;` 안에서 | 성공(경계는 허용) | 0 (rollback) |
| `purpose='other'` | `23514` (`access_grant_purpose_check`) | 0 |
| 지문이 대문자 16진수 64자 / 63자 / 65자 / `g`가 섞인 64자 | 각각 `23514` (`access_grant_credential_digest_check`) | 0 |
| `expires_at = issued_at` 또는 그보다 앞 | `23514` (`access_grant_expires_after_issued_check`) | 0 |

- 4-7-05·4-7-13에서 다른 이유의 오류(시험 프로젝트의 다른 표가 그 회원·계정을 가리키는 경우 등)가 나오면 오류에 적힌 외래키 이름으로 가린다. 이름이 `member_identity_…`·`access_grant_…`로 시작하면 이 SQL의 문제다.

### 4-8. 로그인 상태 만료(`not_after`) — 회원 XA·관리자 XG (2차에서 넣음)

전용 계정 XA·XG의 로그인 상태 **한 행**만 바꾼다. 점검 예외 XE는 5-3-가에서 한다.
준비(SQL 읽기): 두 계정의 로그인 상태를 읽어 `id`(= `<XA_상태번호>`, `<XG_상태번호>` — 각 자격의 `session_id`와 같은 행)와 원래 `not_after`를 적는다. XG는 6자리 확인을 마친 뒤(행의 `aal`이 `aal2`)여야 한다.
```sql
select s.id, s.user_id, s.aal, s.not_after
  from auth.sessions s
 where s.user_id in ('<XA_계정번호>', '<XG_계정번호>');
```
`not_after`는 아래 한 문장으로만 바꾼다(조건에 로그인 상태 번호를 지정 — 매번 `UPDATE 1`이어야 한다. 0이나 2 이상이면 멈춘다). postgres에 `auth.sessions`를 고칠 권한이 없어 실패하면 멈춰 기록한다(작업방은 그 권한을 확인하지 못했다).
```sql
update auth.sessions set not_after = <값> where id = '<XA_상태번호>' and user_id = '<XA_계정번호>';
```
요청 PX = `rpc/zz_b2a_probe`, MX = `GET …/zz_b2a_member_row?select=member_id,note`, WX = `PATCH …/zz_b2a_member_row?member_id=eq.<XA_회원번호>` 본문 `{"note":"<번호>"}`, FX = `PATCH …/zz_b2a_admin_flag?id=eq.1` 본문 `{"flag":"<번호>"}`.

회원 XA:
| 번호 | 그 한 행의 `not_after` | 자격·보낼 요청 | 기대 응답 | 기대하는 실제 행 변화 |
|---|---|---|---|---|
| 4-8-01 | (원래 값 그대로) | KXA: PX | `member_id` = XA의 회원번호 | 없음 |
| 4-8-02 | `pg_catalog.now() - interval '1 hour'` (과거) | KXA: PX | `member_id`·`member_id_text`·`member_email` 모두 null | 없음 |
| 4-8-03 | (과거 그대로) | KXA: MX | `200` `[]` | 없음 |
| 4-8-04 | (과거 그대로) | KXA: WX | `200` `[]` | 0 (XA 행 `note` 그대로) |
| 4-8-05 | `pg_catalog.now() + interval '1 hour'` (미래) | KXA: PX | `member_id` = XA의 회원번호 | 없음 |
| 4-8-06 | (미래 그대로) | KXA: MX / WX | XA의 행 1개 / `200` 1행 | XA 행 `note`만 |
| 4-8-07 | `null` | KXA: PX | `member_id` = XA의 회원번호 | 없음 |
| 4-8-08 | (`null` 그대로) | KXA: MX / WX | XA의 행 1개 / `200` 1행 | XA 행 `note`만 |
| 4-8-09 | 경계(아래 "회원 경계" 묶음, 한 트랜잭션) | SQL | `before_expect_xa` = XA의 회원번호, `at_now_expect_null` = NULL | 없음(rollback) |
| 4-8-10 | 원래 값으로 되돌림 | SQL(읽기): 준비 질의 | XA 행의 `not_after`가 준비 때 적은 값과 같음 | 원래대로 |

관리자 XG(6자리 후 자격 KXG2):
| 번호 | 그 한 행의 `not_after` | 자격·보낼 요청 | 기대 응답 | 기대하는 실제 행 변화 |
|---|---|---|---|---|
| 4-8-11 | (원래 값 그대로) | KXG2: FX | `200` 1행 | `flag` 바뀜 |
| 4-8-12 | 과거 | KXG2: FX | `200` `[]` | 0 (`flag` 그대로) |
| 4-8-13 | (과거 그대로) | KXG2: PX | `is_admin` false | 없음 |
| 4-8-14 | (과거 그대로) | KXG2: `rpc/my_site_access` | `200` `{"allowed": true, "is_admin": false}`(점검 꺼짐이라 `allowed`는 참) | 없음 |
| 4-8-15 | 미래 | KXG2: FX | `200` 1행 | `flag` 바뀜 |
| 4-8-16 | `null` | KXG2: FX | `200` 1행 | `flag` 바뀜 |
| 4-8-17 | 경계(아래 "관리자 경계" 묶음, 한 트랜잭션) | SQL | `before_expect_true` = true, `at_now_expect_false` = false | 없음(rollback) |
| 4-8-18 | 원래 값으로 되돌림 | SQL(읽기): 준비 질의 | XG 행의 `not_after`가 준비 때 적은 값과 같음 | 원래대로 |
| 4-8-19 | (원래 값) | KA: 4-1의 P / KG2″: 4-4의 F | A는 자기 회원번호 / 표식 1행 — 기존 회귀 계정은 영향 없음 | 각자 원래 권한의 변화만 |

회원 경계 — 같은 순간은 만료(한 트랜잭션 안에서는 `now()`가 같다. 끝은 반드시 `rollback`):
```sql
begin;
select pg_catalog.set_config('request.jwt.claims',
  '{"role":"authenticated","aal":"aal1","sub":"<XA_계정번호>","session_id":"<XA_상태번호>"}', true);
select security_private.current_member_id() as before_expect_xa;       -- 흉내 낸 자격이 제대로 읽히는지
update auth.sessions set not_after = pg_catalog.now() where id = '<XA_상태번호>' and user_id = '<XA_계정번호>';
select security_private.current_member_id() as at_now_expect_null;     -- not_after = 지금 → 만료 → NULL
rollback;
```
관리자 경계:
```sql
begin;
select pg_catalog.set_config('request.jwt.claims',
  '{"role":"authenticated","aal":"aal2","sub":"<XG_계정번호>","session_id":"<XG_상태번호>"}', true);
select security_private.is_current_admin() as before_expect_true;
update auth.sessions set not_after = pg_catalog.now() where id = '<XG_상태번호>' and user_id = '<XG_계정번호>';
select security_private.is_current_admin() as at_now_expect_false;
rollback;
```
- 두 묶음은 SQL 편집기(postgres)에서 판정 함수를 직접 부른다. 자격 값은 `set_config`로 흉내 낸 것이다(REST로는 같은 순간을 맞출 수 없어서다). `before_…` 값이 기대와 다르면 흉내 낸 자격이 틀린 것이므로 뒤 값을 판정에 쓰지 않는다.
- 과거로 둔 동안 로그인 서버가 그 로그인 상태를 스스로 지울 수 있는지는 작업방이 모른다. 준비 질의로 행이 그대로 있는지 확인하면서 진행하고, 행이 사라졌으면 그 계정으로 다시 로그인해 새 로그인 상태 번호로 이어 간다(기록에 남긴다).

| SQL로 바꾸는 것 | 자격·요청 | 기대 응답 | 변화 |

**5-3-가. 점검 예외의 로그인 상태 만료 — 전용 계정 XE (2차에서 넣음)**

`phase = 'maintenance'`, `maintenance_id = 101`, XE의 예외 행(만료 없음)이 있는 상태에서 한다. E의 로그인 상태는 건드리지 않는다.
준비(SQL 읽기): XE의 로그인 상태를 읽어 `id`(= `<XE_상태번호>`, KXE의 `session_id`와 같은 행)와 원래 `not_after`를 적는다.
```sql
select s.id, s.user_id, s.aal, s.not_after from auth.sessions s where s.user_id = '<XE_계정번호>';
```
`not_after`는 아래 한 문장으로만 바꾼다(조건에 로그인 상태 번호를 지정 — 매번 `UPDATE 1`이어야 한다. 0이나 2 이상이면 멈춘다).
```sql
update auth.sessions set not_after = <값> where id = '<XE_상태번호>' and user_id = '<XE_계정번호>';
```
| 번호 | 그 한 행의 `not_after` | 자격·보낼 요청 | 기대 응답 | 기대하는 실제 행 변화 |
|---|---|---|---|---|
| 5-3-가-01 | (원래 값 그대로) | KXE: open_row GET | `200` 1행 | 없음 |
| 5-3-가-02 | (원래 값 그대로) | KXE: `rpc/my_site_access` | `200` `{"allowed": true, "is_admin": false}` | 없음 |
| 5-3-가-03 | `pg_catalog.now() - interval '1 hour'` (과거) | KXE: open_row GET | `503` | 0 |
| 5-3-가-04 | (과거 그대로) | KXE: `rpc/my_site_access` | `200` `{"allowed": false, "is_admin": false}` | 없음 |
| 5-3-가-05 | `pg_catalog.now() + interval '1 hour'` (미래) | KXE: open_row GET | `200` 1행 | 없음 |
| 5-3-가-06 | `null` | KXE: open_row GET | `200` 1행 | 없음 |
| 5-3-가-07 | 경계(아래 묶음, 한 트랜잭션) | SQL | `before_expect_true` = true, `at_now_expect_false` = false | 없음(rollback) |
| 5-3-가-08 | 원래 값으로 되돌림 | SQL(읽기): 위 준비 질의 | `not_after`가 준비 때 적은 값과 같음 | 원래대로 |
| 5-3-가-09 | (원래 값) | KE: 5-3-31의 요청 | `200` 1행(E는 영향 없음) | E의 행만 |

경계 — 같은 순간은 만료(한 트랜잭션 안에서는 `now()`가 같다. 끝은 반드시 `rollback`):
```sql
begin;
select pg_catalog.set_config('request.jwt.claims',
  '{"role":"authenticated","aal":"aal1","sub":"<XE_계정번호>","session_id":"<XE_상태번호>"}', true);
select security_private.request_allowed() as before_expect_true;     -- 흉내 낸 자격이 제대로 읽히는지(예외라서 참)
update auth.sessions set not_after = pg_catalog.now() where id = '<XE_상태번호>' and user_id = '<XE_계정번호>';
select security_private.request_allowed() as at_now_expect_false;    -- not_after = 지금 → 만료 → 거짓
rollback;
```
- 이 묶음은 SQL 편집기(postgres)에서 판정 함수를 직접 부른다. 자격 값은 `set_config`로 흉내 낸 것이다(REST로는 같은 순간을 맞출 수 없어서다). `before_expect_true`가 참이 아니면 흉내 낸 자격이 틀린 것이므로 뒤 값을 판정에 쓰지 않는다.
- postgres에 `auth.sessions`를 고칠 권한이 없어 `update`가 실패하면 멈춰 기록한다(작업방은 그 권한을 확인하지 못했다).

## 7단계 — 내리기 → 되돌리기 → 대조 → 재적용 (기준 5)

여기서 쓰는 되돌리기 SQL은 **이 커밋(2차 판)의 것**이다. 1-가에서 쓴 `fa172b2`의 되돌리기 SQL과 섞지 않는다.

| 순서 | 보낼 요청 | 기대 응답 | 기대하는 실제 행 변화 |
|---|---|---|---|
| 7-1 (선택) | 시험 전용 객체가 **남아 있는 채로** 되돌리기 SQL 실행 | 실패(`zz_b2a_*` 표의 정책이 판정 함수에 기대고 있음 — `2BP01`). 한 트랜잭션이라 아무것도 지워지지 않아야 한다 | 0 (`public.site_status`도 그대로 있어야 한다) |
| 7-2 | SQL: `fixtures_down.sql` 전체 | 오류 없음 | `public.zz_b2a_*` 표 4개·함수 4개·정책 사라짐 |
| 7-3 | SQL: 2-1의 질의로 `pgrst.db_pre_request` 항목이 없는지 다시 확인한 뒤, 되돌리기 SQL 전체 | 오류 없음(설정이 남아 있으면 첫 블록에서 `b2a 되돌리기 중단`으로 멈춘다) | `security_private` 스키마와 그 안의 표 5개·함수 8개(도우미 `current_session` 포함), `public.site_status`·`public.my_site_access` 사라짐(표 안의 합성 대응·장부 행도 함께) |
| 7-4 | SQL: 아래 "남은 것 확인" | 세 질의 모두 0행 | — |
| 7-5 | SQL: 남아 있는 합성 회원(`public.users`의 A·B·E·DB·XA)을 지우고, 로그인 서버에서 남아 있는 합성 계정 A·B·N·G·E·DA·DB·DH·XA·XG·XE를 지운다 | 오류 없음 | `public.users` −5, `auth.users` −11, 관련 로그인 상태·인증 앱 등록 0 |
| 7-6 | SQL: 2-2의 지문 질의 전부(지문 8개), 2-1의 질의, 기준 요청 R1~R3 | 2단계에서 저장한 것과 전부 같음 | 없음 |
| 7-7 | SQL: 마이그레이션 파일 전체를 **다시** 실행(재적용) | 오류 없이 끝남(자기 점검 통과) | 3-1과 같은 객체가 다시 생김, `site_state` 1행 |
| 7-8 | `notify pgrst, 'reload schema';` 뒤 K0: `rpc/site_status` | `200`, `phase` `normal`, `maintenance_id` 0 | 없음 |

남은 것 확인:
```sql
select n.nspname from pg_catalog.pg_namespace n where n.nspname = 'security_private';

select n.nspname, p.proname
  from pg_catalog.pg_proc p join pg_catalog.pg_namespace n on n.oid = p.pronamespace
 where (n.nspname = 'public' and p.proname in ('site_status', 'my_site_access'))
    or p.proname like 'zz\_b2a\_%';

select n.nspname, c.relname
  from pg_catalog.pg_class c join pg_catalog.pg_namespace n on n.oid = c.relnamespace
 where c.relname like 'zz\_b2a\_%';
```
- 7-5의 숫자: 합성 회원은 7명(A·B·E·DA·DB·XA·DZ)을 만들었고 4-7에서 DA·DZ를 이미 지웠으므로 5명이 남는다. 합성 계정은 12개(A·B·N·G·E·DA·DB·DG·DH·XA·XG·XE)를 만들었고 4-7에서 DG를 이미 지웠으므로 11개가 남는다. DA는 회원 행만 지워졌고 로그인 계정은 남아 있으니 여기서 지운다.
- 7-5의 순서: **새 판에서는 순서 제약이 없다.** 회원 행을 지우면 그 회원의 연결(`member_identity`)과 그 회원 대상 자격(`access_grant`)이 함께 지워지고, 로그인 계정을 지우면 그 계정의 연결·명부·예외 명단·행위자 자격이 함께 지워진다. 그래서 되돌리기 SQL(7-3) 전에 합성 회원을 지워도 23503(외래키 오류)이 나지 않아야 한다 — 난다면 2차 수정이 적용되지 않은 것이므로 실패로 기록한다. 이 표의 순서(7-3 뒤에 7-5)대로 하면 그때는 새 표가 이미 없다.
- 7-6의 (사)(계정·로그인 상태 수)는 7-5까지 끝낸 뒤에 2단계와 같아야 한다.
- **7-7 뒤의 최종 상태**(재적용한 채로 둘지, 되돌리기 SQL을 한 번 더 실행해 비울지)는 계약서에 정해져 있지 않다. 브레인방이 정한다(1차 실행에서는 재적용한 채로 두었다). 재적용한 채로 두는 경우 2-2의 지문은 그대로여야 하고(새 객체는 지문에서 빠져 있다), 공통 확인 설정은 원복된 상태여야 한다.
- "훅 재적용"(6-2)과 "마이그레이션 재적용"(7-7)은 서로 다른 확인이다. 둘 다 한다.

## 이 절차가 확인하지 않는 것

- 실제 화면·실제 업무(대리·책의 발급과 중계, 가입·비밀번호)는 이 계약 범위 밖이다. `resolve_grant`의 1행은 장부 확인일 뿐 실제 자료 접근 승인이 아니다.
- 파일(Storage) 규칙과 Edge 함수에서의 같은 판정은 이 절차에 없다(공통 확인은 PostgREST에만 걸린다).
- DB 안에서 도는 예약·트리거·직접 SQL은 공통 확인을 거치지 않는다.
- `my_site_access`의 `Cache-Control` 머리글은 API 주소에 직접 보낸 요청에서만 확인됐다. 브라우저가 쓰는 실제 배포 경로(중간 저장소 포함)에서는 아직 보지 않았다.
- 로그인 서버가 `not_after`를 실제로 어떤 경우에 채우는지(상태 유효 기간 설정 등)는 이 절차가 보지 않는다. 여기서는 그 열의 값을 직접 바꿨을 때 판정이 따라가는지만 본다.
- 이 수정은 기존 탈퇴 파기 예약의 오류(text = uuid)를 고치지 않는다(묶음 6). 4-7은 "새 표가 회원·계정 삭제를 막지 않는다"까지만 본다.
- 동시에 많은 요청이 올 때의 응답 시간은 재지 않는다. 로그인 상태 판정 도우미는 jsonb 변환에 예외 처리 블록을 쓰므로 호출마다 작은 비용이 있다 — 정책에서는 `(select security_private.current_member_id())`처럼 감싸 한 번만 계산되게 쓰는 것을 권한다(`zz_b2a_member_row`가 그 형태, `zz_b2a_admin_flag`는 직접 부르는 형태다).
- 자격의 유효 시간(60분) 내내 같은 결과인지는 재지 않는다.

## 1차 실행(커밋 `fa172b2`, 2026-11-23 브레인방)에서 이미 확인된 것 — 이 문서의 기대값 근거

- 1차 판 마이그레이션은 첫 실행에 자기 점검까지 통과했다(PostgreSQL 17.12). **2차 판의 마이그레이션·자기 점검·되돌리기 SQL은 아직 실행된 적이 없다.**
- 점검 거절의 응답 모양: `503`, 본문 `code` `PT503`·`message` `maintenance`.
- `my_site_access`: 로그인 회원 `200` + `Cache-Control: private, no-store`(API 직접 요청), 비로그인 `401`, 서비스 열쇠 `403`.
- 2-2의 지문 질의는 전부 실행됐다(문법 오류 없음). 기준 요청 R3는 익명이 부를 수 있는 읽기 전용 rpc가 없어 익명으로 읽히는 표 조회로 대신했다.
- 이 문서에서 **아직 한 번도 실행되지 않은 부분**: 1-가, 3-7·3-8, 4-7·4-8·4-9, 5-3-가, 그리고 5-3·5-4를 항목별로 나눈 기록 방식.
